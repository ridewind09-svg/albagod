                area.innerHTML = `<div class="write-form"><h3 style="margin-bottom:15px;">글쓰기</h3><input type="text" id="${type}Title" placeholder="제목을 입력하세요"><textarea id="${type}Content" placeholder="내용을 입력하세요"></textarea><button class="btn-write" onclick="writePost('${type}')">작성 완료</button></div>`;
            }
        }

        function handleSignup() {
            const id = document.getElementById('signupId').value.trim();
            const pw = document.getElementById('signupPw').value.trim();
            const pwCheck = document.getElementById('signupPwCheck').value.trim();
            const nickname = document.getElementById('signupNickname').value.trim();
            const name = document.getElementById('signupName').value.trim();
            const phone = document.getElementById('signupPhone').value.trim();

            if(!id || !pw || !nickname || !name || !phone) { alert('모든 항목을 입력해주세요.'); return; }
            if(pw !== pwCheck) { alert('비밀번호가 일치하지 않습니다.'); return; }
            if(users.find(u => u.id === id)) { alert('이미 존재하는 아이디입니다.'); return; }

            users.push({ id, pw, nickname, name, phone, provider: '일반', scraps: [] });
            alert('회원가입이 완료되었습니다. 로그인해주세요.');
            showPage('login', document.querySelector('.auth-menu a'));
        }

        function handleLogin() {
            const id = document.getElementById('loginId').value.trim();
            const pw = document.getElementById('loginPw').value.trim();
            const user = users.find(u => u.id === id && u.pw === pw);
            if(!user) { alert('아이디 또는 비밀번호가 일치하지 않습니다.'); return; }
            currentUser = user;
            if(!currentUser.scraps) currentUser.scraps = []; 
            alert(`${user.nickname}님 환영합니다!`);
            updateAuthMenu();
            showPage('home', document.querySelector('.nav-menu a'));
        }

        function openSocialModal(provider) {
            tempSocialProvider = provider;
            document.getElementById('socialProviderName').innerText = provider;
            document.getElementById('socialModal').style.display = 'flex';
        }

        function confirmSocialLogin() {
            const nickname = document.getElementById('socialNicknameInput').value.trim();
            if(!nickname) { alert('닉네임을 입력해주세요.'); return; }

            currentUser = {
                id: `${tempSocialProvider}_user_${Math.floor(Math.random() * 10000)}`,
                name: `${tempSocialProvider} 유저`,
                nickname: nickname,
                phone: '소셜 로그인 제공',
                provider: tempSocialProvider,
                scraps: []
            };
            users.push(currentUser);
            closeModal('socialModal');
            alert(`${tempSocialProvider} 로그인 성공!`);
            updateAuthMenu();
            showPage('home', document.querySelector('.nav-menu a'));
        }

        function handleLogout() {
            currentUser = null;
            alert('로그아웃 되었습니다.');
            updateAuthMenu();
            showPage('home', document.querySelector('.nav-link'));
        }

        function updateAuthMenu() {
            const authMenu = document.getElementById('authMenu');
            const adminLink = document.getElementById('adminMenuLink');
            if(currentUser) {
                authMenu.innerHTML = `<a style="background:#4a4a60; cursor:default;">${currentUser.nickname}님</a><a class="logout-btn" onclick="handleLogout()">로그아웃</a>`;
                adminLink.style.display = currentUser.id === 'admin' ? 'block' : 'none';
            } else {
                authMenu.innerHTML = `<a onclick="showPage('login', this)">로그인</a><a onclick="showPage('signup', this)">회원가입</a>`;
                adminLink.style.display = 'none';
            }
        }

        function loadMypage() {
            const mypageContent = document.getElementById('mypageContent');
            if(!currentUser) {
                mypageContent.innerHTML = `<p style="text-align:center; color:#666;">로그인이 필요한 페이지입니다.</p>`;
                return;
            }

            let myPosts = [];
            ['semi', 'other', 'free'].forEach(type => {
                posts[type].forEach((p, idx) => {
                    if(currentUser.id === p.authorId) myPosts.push({ type, index: idx, post: p });
                });
            });

            let scrapPosts = [];
            if(currentUser.scraps) {
                ['semi', 'other', 'free'].forEach(type => {
                    posts[type].forEach((p, idx) => {
                        if(currentUser.scraps.includes(p.id)) scrapPosts.push({ type, index: idx, post: p });
                    });
                });
            }

            let myPostsHtml = myPosts.length === 0 ? `<p style="color:#999;">작성한 글이 없습니다.</p>` : 
                myPosts.map(m => `<div class="post-item" onclick="viewPost('${m.type}', ${m.index})"><h3>${m.post.title}</h3><div class="meta">작성일: ${m.post.date} <span class="view-btn">[보기]</span></div></div>`).join('');

            let scrapPostsHtml = scrapPosts.length === 0 ? `<p style="color:#999;">스크랩한 글이 없습니다.</p>` : 
                scrapPosts.map(s => `<div class="post-item" onclick="viewPost('${s.type}', ${s.index})"><h3>${s.post.title}</h3><div class="meta">작성자: ${s.post.author} | ${s.post.date} <span class="view-btn">[보기]</span></div></div>`).join('');

            mypageContent.innerHTML = `
                <h2 style="margin-bottom:20px;">${currentUser.nickname}님의 마이페이지</h2>
                <div class="mypage-info">
                    <div class="info-row"><div class="info-label">아이디</div><div class="info-value">${currentUser.id}</div></div>
                    <div class="info-row"><div class="info-label">닉네임</div><div class="info-value">${currentUser.nickname}</div></div>
                    <div class="info-row"><div class="info-label">이름</div><div class="info-value">${currentUser.name}</div></div>
                    <div class="info-row"><div class="info-label">연락처</div><div class="info-value">${currentUser.phone}</div></div>
                    <div class="info-row"><div class="info-label">가입 방식</div><div class="info-value">${currentUser.provider}</div></div>
                </div>
                <div class="mypage-section">
                    <h3>📝 내가 쓴 게시글</h3>
                    <div class="post-list">${myPostsHtml}</div>
                </div>
                <div class="mypage-section">
                    <h3>⭐ 나의 스크랩 목록</h3>
                    <div class="post-list">${scrapPostsHtml}</div>
                </div>
            `;
        }

            tbody.innerHTML = users.map(u => `
                <tr>
                    <td>${u.id}</td><td>${u.nickname}</td><td>${u.name}</td><td>${u.phone}</td><td>${u.provider}</td>
                    <td>${u.id === 'admin' ? '<span style="color:#999;">삭제 불가</span>' : `<button class="btn-action btn-delete" onclick="deleteUser('${u.id}')">삭제</button>`}</td>
                </tr>
            `).join('');
        }

        function deleteUser(userId) {
            if(confirm(`${userId} 님을 정말 삭제하시겠습니까?`)) {
                users = users.filter(u => u.id !== userId);
                alert('회원이 삭제되었습니다.');

            }
        }

        function writePost(type) {
            const titleInput = document.getElementById(`${type}Title`);
            const contentInput = document.getElementById(`${type}Content`);
            if(titleInput.value.trim() === "" || contentInput.value.trim() === "") { alert('내용을 입력해주세요.'); return; }

            const newPost = { id: postIdCounter++, title: titleInput.value, content: contentInput.value, author: currentUser.nickname, authorId: currentUser.id, date: new Date().toLocaleString() };
            posts[type].unshift(newPost);
            titleInput.value = ""; contentInput.value = "";
            alert('글이 작성되었습니다.');
            renderPosts();
        }

        function renderPosts() {
            ['semi', 'other', 'free'].forEach(type => {
                const list = document.getElementById(`${type}List`);
                if (posts[type].length === 0) list.innerHTML = `<p style="color:#999;">등록된 글이 없습니다.</p>`;
                else list.innerHTML = posts[type].map((p, index) => `<div class="post-item" onclick="viewPost('${type}', ${index})"><h3>${p.title}</h3><p>${p.content.length > 50 ? p.content.substring(0, 50) + '...' : p.content}</p><div class="meta">작성자: ${p.author} | ${p.date} <span class="view-btn">[클릭하여 보기]</span></div></div>`).join('');
            });
            renderMainList('mainSemiList', 'semi');
            renderMainList('mainOtherList', 'other');
        }

        function renderMainList(elementId, type) {
            const list = document.getElementById(elementId);
            const latestPosts = posts[type].slice(0, 3);
            if (latestPosts.length === 0) list.innerHTML = `<p style="color:#999;">아직 등록된 글이 없습니다.</p>`;
            else list.innerHTML = latestPosts.map((p, index) => `<div class="post-item" onclick="viewPost('${type}', ${index})"><h3>${p.title}</h3><div class="meta">작성자: ${p.author} | ${p.date}</div></div>`).join('');
        }

        function viewPost(type, index) {
            const post = posts[type][index];
            document.getElementById('modalTitle').innerText = post.title;
            document.getElementById('modalAuthor').innerText = post.author;
            document.getElementById('modalDate').innerText = post.date;
            document.getElementById('modalContent').innerText = post.content;

            const actionsDiv = document.getElementById('modalActions');
            const moveArea = document.getElementById('modalMoveArea');
            actionsDiv.innerHTML = '';
            moveArea.innerHTML = '';
            moveArea.style.display = 'none';

            if(currentUser) {
                let html = '';
                if(currentUser.id === post.authorId || currentUser.id === 'admin') {
                    html += `<button class="btn-action btn-edit" onclick="editPost('${type}', ${index})">수정하기</button>`;
                    html += `<button class="btn-action btn-delete" onclick="deletePost('${type}', ${index})">삭제하기</button>`;
                }
                
                if(currentUser.id !== post.authorId) {
                    const isScrapped = currentUser.scraps.includes(post.id);
                    html += `<button class="btn-action btn-scrap" onclick="toggleScrap('${type}', ${index})">${isScrapped ? '스크랩 취소' : '스크랩하기'}</button>`;
                }
                actionsDiv.innerHTML = html;

                if(currentUser.id === 'admin') {
                    actionsDiv.innerHTML += `<button class="btn-action btn-rec" onclick="addPostToRec('${type}', ${index})">추천 알바 등록</button>`;
                    
                    moveArea.style.display = 'block';
                    moveArea.innerHTML = `
                        <label style="font-weight:700; margin-right:10px;">게시판 이동 (관리자):</label>
                        <select id="moveSelect">
                            <option value="">게시판 선택</option>
                            <option value="semi">반도체현장공고</option>
                            <option value="other">기타알바공고</option>
                            <option value="free">자유게시판</option>
                        </select>
                        <button class="btn-action btn-move" onclick="movePost('${type}', ${index})">이동하기</button>
                    `;
                }
            }

            document.getElementById('postModal').style.display = 'flex';
        }

        function toggleScrap(type, index) {
            const post = posts[type][index];
            const scrapIndex = currentUser.scraps.indexOf(post.id);
            if(scrapIndex > -1) {
                currentUser.scraps.splice(scrapIndex, 1);
                alert('스크랩이 취소되었습니다.');
            } else {
                currentUser.scraps.push(post.id);
                alert('스크랩되었습니다. 마이페이지에서 확인하세요!');
            }
            viewPost(type, index);
        }

        function addPostToRec(type, index) {
            const post = posts[type][index];
            recommendedJobs.unshift({ id: recJobIdCounter++, title: post.title, info: post.content.substring(0, 20) + '...' });
            alert('오늘의 추천 알바에 등록되었습니다!');
            renderHomeRecJobs();
            closeModal('postModal');
        }

        function movePost(type, index) {
            const newType = document.getElementById('moveSelect').value;
            if(!newType) { alert('이동할 게시판을 선택하세요.'); return; }
            if(newType === type) { alert('같은 게시판입니다.'); return; }

            const post = posts[type].splice(index, 1)[0]; 
            posts[newType].unshift(post); 
            
            alert('게시판이 이동되었습니다.');
            renderPosts();
            closeModal('postModal');
        }

        function editPost(type, index) {
            const post = posts[type][index];
            document.getElementById('modalTitle').innerHTML = `<input type="text" id="editTitle" value="${post.title}" style="width:100%; padding:10px; border:1px solid #ccc; border-radius:5px;">`;
            document.getElementById('modalContent').innerHTML = `<textarea id="editContent" style="width:100%; height:200px; padding:10px; border:1px solid #ccc; border-radius:5px;">${post.content}</textarea>`;
            document.getElementById('modalActions').innerHTML = `<button class="btn-action btn-edit" onclick="savePost('${type}', ${index})">저장하기</button><button class="btn-action" style="background:#ccc; color:#333;" onclick="viewPost('${type}', ${index})">취소</button>`;
            document.getElementById('modalMoveArea').style.display = 'none';
        }

        function savePost(type, index) {
            const newTitle = document.getElementById('editTitle').value.trim();
            const newContent = document.getElementById('editContent').value.trim();
            if(!newTitle || !newContent) { alert('제목과 내용을 모두 입력해주세요.'); return; }
            posts[type][index].title = newTitle;
            posts[type][index].content = newContent;
            posts[type][index].date = `${new Date().toLocaleString()} (수정됨)`;
            alert('글이 수정되었습니다.');
            renderPosts();
            viewPost(type, index);
        }

        function deletePost(type, index) {
            if(confirm('정말 삭제하시겠습니까?')) {
                posts[type].splice(index, 1);
                alert('삭제되었습니다.');
                renderPosts();
                closeModal('postModal');
            }
        }

        function closeModal(modalId) { document.getElementById(modalId).style.display = 'none'; }
        window.onclick = function(event) { if (event.target.id === 'postModal') closeModal('postModal'); if (event.target.id === 'socialModal') closeModal('socialModal'); if (event.target.id === 'chatModal') closeModal('chatModal'); }

        function addRecJob() {
            const title = document.getElementById('recJobTitle').value.trim();
            const info = document.getElementById('recJobInfo').value.trim();
            if(!title || !info) { alert('제목과 정보를 모두 입력해주세요.'); return; }
            recommendedJobs.unshift({ id: recJobIdCounter++, title, info });
            document.getElementById('recJobTitle').value = '';
            document.getElementById('recJobInfo').value = '';
            renderAdminRecJobs();
            renderHomeRecJobs();
        }

        function deleteRecJob(id) {
            if(confirm('추천 알바에서 삭제하시겠습니까?')) {
                recommendedJobs = recommendedJobs.filter(j => j.id !== id);
                renderAdminRecJobs();
                renderHomeRecJobs();
            }
        }

        function renderAdminRecJobs() {
            const list = document.getElementById('adminRecJobList');
            if(recommendedJobs.length === 0) list.innerHTML = `<p style="color:#999;">등록된 추천 알바가 없습니다.</p>`;
            else list.innerHTML = recommendedJobs.map(j => `<div class="job-card"><div class="job-title">${j.title}</div><div class="job-info">${j.info}</div><button class="btn-action btn-delete" style="margin-top:10px;" onclick="deleteRecJob(${j.id})">삭제</button></div>`).join('');
        }

        function renderHomeRecJobs() {
            const list = document.getElementById('homeJobList');
            if(recommendedJobs.length === 0) list.innerHTML = `<p style="color:#999;">등록된 추천 알바가 없습니다.</p>`;
            else list.innerHTML = recommendedJobs.map(j => `<div class="job-card"><div class="job-title">${j.title}</div><div class="job-info">${j.info}</div></div>`).join('');
        }

        renderHomeRecJobs();
        renderPosts();

        function openChatModal() {
            const mainScreen = document.getElementById('chatScreen');
            const modalScreen = document.getElementById('chatModalScreen');
            modalScreen.innerHTML = mainScreen.innerHTML;
            modalScreen.scrollTop = modalScreen.scrollHeight;
            document.getElementById('chatModal').style.display = 'flex';

        }

        function sendMessage(isModal = false) {
            const inputId = isModal ? 'chatModalInput' : 'chatInput';
            const input = document.getElementById(inputId);
            const message = input.value.trim();
            if (message !== "") {
                const author = currentUser ? currentUser.nickname : '나';
                const newMessageHTML = `<div class="chat-message"><span class="user">${author}:</span> ${message}</div>`;
                
                document.getElementById('chatScreen').insertAdjacentHTML('beforeend', newMessageHTML);
                document.getElementById('chatModalScreen').insertAdjacentHTML('beforeend', newMessageHTML);
                
                document.getElementById('chatScreen').scrollTop = document.getElementById('chatScreen').scrollHeight;
                document.getElementById('chatModalScreen').scrollTop = document.getElementById('chatModalScreen').scrollHeight;
                
                input.value = "";
            }
        }
        function handleKeyPress(event) { if (event.key === 'Enter') sendMessage(false); }
        function handleModalKeyPress(event) { if (event.key === 'Enter') sendMessage(true); }
